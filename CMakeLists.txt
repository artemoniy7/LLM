cmake_minimum_required(VERSION 3.20)
project(rp_llm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(rp_llm src/main.cpp)
target_include_directories(rp_llm PRIVATE include)
target_compile_options(rp_llm PRIVATE -O2 -Wall -Wextra -Wpedantic)
if(MSVC)
  target_compile_options(rp_llm PRIVATE /O2)
endif()
